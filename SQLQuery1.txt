-- 🔹 STEP 1: Ensure Admin role exists
IF NOT EXISTS (SELECT 1 FROM AspNetRoles WHERE Name = 'Admin')
BEGIN
    INSERT INTO AspNetRoles (Id, Name, NormalizedName, ConcurrencyStamp) 
    VALUES (NEWID(), 'Admin', 'ADMIN', NEWID());
    PRINT '✅ Created Admin role';
END

-- 🔹 STEP 2: Find the user by email
DECLARE @UserId NVARCHAR(450) = (SELECT Id FROM AspNetUsers WHERE Email = 'admin@cca.com');

IF @UserId IS NULL
BEGIN
    PRINT '❌ User admin@cca.com not found. Please register the account first.';
    RETURN;
END

-- 🔹 STEP 3: Get Admin Role ID
DECLARE @AdminRoleId NVARCHAR(450) = (SELECT Id FROM AspNetRoles WHERE Name = 'Admin');

-- 🔹 STEP 4: Add user to Admin role (if not already)
IF NOT EXISTS (SELECT 1 FROM AspNetUserRoles WHERE UserId = @UserId AND RoleId = @AdminRoleId)
BEGIN
    INSERT INTO AspNetUserRoles (UserId, RoleId) VALUES (@UserId, @AdminRoleId);
    PRINT '✅ admin@cca.com is now an Admin!';
END
ELSE
BEGIN
    PRINT '⚠️ admin@cca.com is already an Admin.';
END

-- 🔹 STEP 5: Verify
SELECT 
    u.Email, 
    u.FullName, 
    r.Name AS Role
FROM AspNetUsers u
JOIN AspNetUserRoles ur ON u.Id = ur.UserId
JOIN AspNetRoles r ON ur.RoleId = r.Id
WHERE u.Email = 'admin@cca.com';
